import sys import time class CyberSecurityAcademy: def __init__(self): self.score = 0 self.total_questions = 0 self.user_name = "" def clear_screen(self): print("\n" * 3) def print_header(self, title): print("=" * 60) print(f" {title.center(58)} ") print("=" * 60) def start(self): self.print_header("مرحباً بك في أكاديمية الأمن السيبراني الشاملة") self.user_name = input("[-] الرجاء إدخال اسمك للبدء: ") print(f"\n[+] أهلاً بك يا {self.user_name}. جهز نفسك لرحلة احترافية!") time.sleep(1) self.main_menu() def main_menu(self): while True: self.clear_screen() self.print_header("القائمة الرئيسية للمنصة") print("1. 📚 المسارات التعليمية (المحتوى النظري والعملي)") print("2. 🛠️ المختبر والأدوات العملية (التشفير وفحص الشبكات)") print("3. 📝 نظام الاختبارات والتقييم (Quiz)") print("4. 📊 عرض الإنجازات والنتيجة الحالية") print("5. ❌ الخروج من التطبيق") print("=" * 60) choice = input("[?] اختر رقم القسم الذي تريد الدخول إليه: ") if choice == '1': self.education_tracks() elif choice == '2': self.practical_lab() elif choice == '3': self.start_interactive_quiz() elif choice == '4': self.show_achievements() elif choice == '5': print(f"\n[+] شكراً لاستخدامك الأكاديمية يا {self.user_name}. بالتوفيق في مسيرتك!") sys.exit() else: print("\n[!] اختيار غير صحيح، حاول مجدداً.") time.sleep(1) # ---------------- 1. القسم التعليمي ---------------- def education_tracks(self): while True: self.clear_screen() self.print_header("📚 مسارات تعلم الأمن السيبراني") print("1. أساسيات الأمن السيبراني (CIA Triad)") print("2. ثغرات الويب الشهيرة (OWASP Top 10)") print("3. أمن الشبكات والبروتوكولات (Network Security)") print("4. علم التشفير (Cryptography)") print("5. العودة للقائمة الرئيسية") print("=" * 60) track = input("[?] اختر المسار التعليمي لقراءته: ") if track == '1': self.show_lesson("أساسيات الأمن السيبراني", "مثلث الـ CIA هو أساس الأمن السيبراني ويتكون من:\n" "1. السرية (Confidentiality): ضمان ألا يطلع على البيانات إلا المصرح لهم فقط. (مثل التشفير)\n" "2. السلامة (Integrity): ضمان عدم تعديل البيانات أو تزويرها أثناء النقل. (مثل الـ Hashing)\n" "3. التوافر (Availability): ضمان أن تكون الأنظمة والخدمات متاحة للمستخدمين دائماً. (مكافحة هجمات DoS)") elif track == '2': self.show_lesson("ثغرات الويب (OWASP Top 10)", "أخطر ثغرات تطبيقات الويب تشمل:\n" "1. SQL Injection (SQLi): حقن أوامر برمجية خبيثة في قواعد البيانات لسرقة المعلومات.\n" "2. Cross-Site Scripting (XSS): حقن سكريبتات (JavaScript) في صفحات يراها مستخدمون آخرون لسرقة الجلسات (Cookies).\n" "3. Broken Authentication: ضعف نظام تسجيل الدخول يتيح للمخترقين تخمين كلمة المرور بسهولة.") elif track == '3': self.show_lesson("أمن الشبكات", "الشبكات هي الممر الأساسي للبيانات، وأهم المفاهيم فيها:\n" "1. جدار الحماية (Firewall): جهاز أو برنامج يراقب حركة مرور الشبكة ويمنع الاتصالات غير المصرح بها.\n" "2. هجمات حجب الخدمة (DDoS): إغراق السيرفر بطلبات وهمية ضخمة من أجهزة متعددة لتعطيله عن العمل.\n" "3. بروتوكول HTTPS: النسخة الآمنة من HTTP وتقوم بتشفير البيانات بين المتصفح والسيرفر باستخدام TLS/SSL.") elif track == '4': self.show_lesson("علم التشفير (Cryptography)", "ينقسم التشفير إلى نوعين رئيسيين:\n" "1. التشفير المتماثل (Symmetric): استخدام نفس المفتاح للتشفير وفك التشفير (مثل خوارزمية AES).\n" "2. التشفير غير المتماثل (Asymmetric): استخدام مفتاحين؛ مفتاح عام (Public) للتشفير ومفتاح خاص (Private) لفك التشفير (مثل RSA).") elif track == '5': break else: print("[!] اختيار غير متاح.") time.sleep(1) def show_lesson(self, title, content): self.clear_screen() self.print_header(title) print(content) print("=" * 60) input("\n[+] اضغط Enter للعودة إلى قائمة المسارات...") # ---------------- 2. المختبر العملي ---------------- def practical_lab(self): while True: self.clear_screen() self.print_header("🛠️ المختبر العملي والأدوات") print("1. تشغيل أداة محاكاة فحص المنافذ (Port Scanner)") print("2. تشغيل أداة تشفير النصوص (Caesar Cipher)") print("3. العودة للقائمة الرئيسية") print("=" * 60) tool = input("[?] اختر الأداة التي تريد تجربتها: ") if tool == '1': self.port_scanner_tool() elif tool == '2': self.caesar_cipher_tool() elif tool == '3': break else: print("[!] اختيار خاطئ.") time.sleep(1) def port_scanner_tool(self): self.clear_screen() self.print_header("محاكي فحص المنافذ (Port Scanner Simulator)") ip = input("[-] أدخل عنوان الـ IP المراد فحصه (مثال: 192.168.1.1): ") print(f"\n[*] جاري بدء الفحص على {ip}...") time.sleep(1) # محاكاة فحص لبعض المنافذ الشهيرة ports = {21: "FTP (Open)", 22: "SSH (Open)", 80: "HTTP (Open)", 443: "HTTPS (Open)", 3389: "RDP (Closed)"} for port, status in ports.items(): print(f"[+] Checking Port {port}... Found: {status}") time.sleep(0.4) print("\n[+] اكتمل الفحص الأمني بنجاح!") input("\nاضغط Enter للعودة...") def caesar_cipher_tool(self): self.clear_screen() self.print_header("أداة تشفير سيزر (Caesar Cipher Tool)") text = input("[-] أدخل النص (باللغة الإنجليزية فقط): ") try: shift = int(input("[-] أدخل مفتاح التشفير (رقم من 1-25): ")) except ValueError: print("[!] يجب إدخال رقم. تم إلغاء العملية.") time.sleep(1) return encrypted = "" for char in text: if char.isalpha(): ascii_offset = 65 if char.isupper() else 97 encrypted += chr((ord(char) - ascii_offset + shift) % 26 + ascii_offset) else: encrypted += char print(f"\n[+] النص المشفر هو: {encrypted}") input("\nاضغط Enter للعودة...") # ---------------- 3. نظام الاختبارات ---------------- def start_interactive_quiz(self): self.clear_screen() self.print_header("📝 بدء الاختبار الأكاديمي الشامل") print("سيتكون الاختبار من أسئلة قوية لقياس مستواك الحقيقي.") input("[!] اضغط Enter عندما تكون مستعداً لبدء السؤال الأول...") # تفاصيل الأسئلة: (السؤال، الاختيارات، الإجابة الصحيحة الرقمية) questions = [ { "question": "ما هو البروتوكول المسؤول عن تشفير حركة المرور بين المتصفح والموقع؟", "options": ["1. Cross-Site Scripting (XSS)", "4. Brute Force"], "answer": "3" }, { "question": "في التشفير غير المتماثل (Asymmetric)، أي مفتاح يُستخدم لتشفيير البيانات؟", "options": ["1. المفتاح الخاص (Private Key)", "2. المفتاح العام (Public Key)", "3. مفتاح الجلسة (Session Key)", "4. لا شيء مما سبق"], "answer": "2" }, { "question": "هجمة الـ DDoS Attack تضرب أي ضلع من أضلاع مثلث الـ CIA للأمن؟", "options": ["1. Confidentiality (السرية)", "2. Integrity (السلامة)", "3. Availability (التوافر)", "4. Non-repudiation"], "answer": "3" }, { "question": "ما هي الوظيفة الأساسية لجدار الحماية (Firewall) في الشبكة؟", "options": ["1. تشفير الملفات الشخصية", "2. تصفية وفحص حزم البيانات الواردة والصادرة", "3. تسريع سرعة الإنترنت", "4. توليد كلمات مرور قوية"], "answer": "2" } ] local_score = 0 for i, q in enumerate(questions, 1): self.clear_screen() print(f"السؤال رقم ({i}/{len(questions)}):") print(f"🔹 {q['question']}\n") for option in q['options']: print(option) print("-" * 40) user_ans = input("[?] اختر رقم الإجابة الصحيحة: ").strip() if user_ans == q['answer']: print("\n✅ إجابة صحيحة! أحسنت.") local_score += 1 else: correct_idx = int(q['answer']) - 1 print(f"\n❌ إجابة خاطئة. الإجابة الصحيحة هي: {q['options'][correct_idx]}") time.sleep(2) # تحديث السجل الإجمالي للمستخدم self.score += local_score self.total_questions += len(questions) self.clear_screen() self.print_header("نتائج الاختبار الحالي") print(f"لقد أجبت بشكل صحيح على {local_score} من أصل {len(questions)} أسئلة.") percentage = (local_score / len(questions)) * 100 print(f"نسبة نجاحك في هذا الاختبار: {percentage}%")
