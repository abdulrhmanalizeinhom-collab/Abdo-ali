cd PhoneSploit-Pro && pip install -r requirements.txt && python phonesploitpro.py

أداة Wifite2 Wi-Fi Automated

pkg install python git iw wireless-tools tshark -y && git clone https://github.com
cd wifite2 && sudo python3 Wifite.py
